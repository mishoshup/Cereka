                // ...
                
                case compiler::Op::PLAY_BGM:
                    impl.audio.PlayBGM(ins.a);
                    si.pc++;
                    continue;
                    
                case compiler::Op::STOP_BGM:
                    impl.audio.StopBGM();
                    si.pc++;
                    continue;
                    
                case compiler::Op::PLAY_SFX:
                    impl.audio.PlaySFX(ins.a);
                    si.pc++;
                    continue;

                    si.pc++;
                    continue;

                    si.pc++;

                    return;

                    ctx.changeState(CerekaState::Finished);
