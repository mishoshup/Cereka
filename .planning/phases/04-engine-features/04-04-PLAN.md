    ```cpp
    #pragma once
    // rollback_manager.hpp — Configurable state ring-buffer snapshots
    //
    // Captures full engine state (dialogue, scene, variables, PC, etc.) at
    // every dialogue advance or menu choice. Restore is instant — reads from
    // in-memory buffer, no file I/O.
    
    #include "cereka_save_data.hpp"  // SerializableSaveData (reuse for snapshot schema)
    #include <cstddef>
    #include <string>
    #include <vector>
    
    namespace cereka {
    
    class CerekaImpl;  // forward declaration
    
    class RollbackManager {
    public:
        explicit RollbackManager(size_t capacity = 200)
            : buffer_(capacity), capacity_(capacity) {}
    
        // Set capacity (on init from config). 0 = rollback disabled.
        void setCapacity(size_t cap);
        
        // Capture current state. Called after each SAY/NARRATE/MENU dispatch.
        void capture(const CerekaImpl &impl);
        
        // Restore to the most recent snapshot. Returns false if no snapshots.
        // On success: state machine is reset, overlays cleared.
        bool restore(CerekaImpl &impl);
        
        // Go to a specific history index (from HistoryState click).
        bool goTo(CerekaImpl &impl, size_t index);
        
        // Get dialogue history texts for UI display (newest first).
        std::vector<std::string> historyTexts() const;
        
        bool canRollback() const { return count_ > 0 && enabled_; }
        void clear() { head_ = count_ = 0; }
        size_t capacity() const { return capacity_; }
        size_t count() const { return count_; }
        
    private:
        std::vector<SerializableSaveData> buffer_;
        size_t capacity_;
        size_t head_ = 0;
        size_t count_ = 0;
        bool enabled_ = true;
        
        // Dialogue text for each snapshot (not serialized in save data, tracked separately)
        std::vector<std::string> dialogueTexts_;
        
        size_t prevIndex() const;  // index of most recent snapshot
    };
    
    }  // namespace cereka
    ```

    ```cpp
    #include "cereka_rollback_manager.hpp"
    #include "cereka_engine_impl.hpp"
    
    namespace cereka {
    
    void RollbackManager::setCapacity(size_t cap) {
        capacity_ = cap;
        buffer_.resize(cap);
        if (cap == 0) {
            enabled_ = false;
            clear();
        } else {
            enabled_ = true;
            if (count_ > cap) count_ = cap;
        }
    }
    
    void RollbackManager::capture(const CerekaImpl &impl) {
        if (capacity_ == 0 || !enabled_) return;
        
        auto &slot = buffer_[head_];
        slot = {};  // reset to defaults
        
        const auto &si = impl.scriptInterpreter;
        const auto &scene = impl.scene;
        const auto &dialogue = impl.dialogue;
        
        // Capture script state
        slot.programCounter = si.pc;
        slot.callStack = si.callStack;
        slot.variables = si.variables;
        slot.numVariables = si.numVariables;
        slot.skipMode = si.skipMode;
        slot.skipDepth = si.skipDepth;
        
        // Capture scene state
        slot.background = scene.BgPath();
        slot.characters.clear();
        for (const auto &[id, entry] : scene.Characters()) {
            auto &ch = slot.characters.emplace_back();
            ch.id = id;
            auto it = scene.CharPaths().find(id);
            ch.file = (it != scene.CharPaths().end()) ? it->second : "";
            ch.position = "center";  // approximate; position reconstructed from xNorm
        }
        
        // Capture audio state
        slot.bgm = impl.audio.BgmPath();
        
        // Capture dialogue state
        slot.speaker = dialogue.Speaker();
        slot.name = dialogue.Name();
        slot.text = dialogue.Text();
        slot.displayedChars = dialogue.DisplayedChars();
        
        // Capture machine state
        slot.state = CerekaStateMachine::stateLabel(si.scriptFinished ? CerekaState::Finished
                                          : (impl.m_stateMachine.hasOverlays()
                                              ? impl.m_stateMachine.effectiveState()
                                              : impl.m_stateMachine.currentType()));
        
        // Record dialogue text for history UI
        dialogueTexts_.resize(buffer_.size());
        dialogueTexts_[head_] = dialogue.Text();
        
        head_ = (head_ + 1) % capacity_;
        if (count_ < capacity_) ++count_;
    }
    
    bool RollbackManager::restore(CerekaImpl &impl) {
        if (count_ == 0) return false;
        return goTo(impl, prevIndex());
    }
    
    bool RollbackManager::goTo(CerekaImpl &impl, size_t index) {
        if (index >= capacity_ || count_ == 0) return false;
        
        // Find the actual buffer index for the history position
        // historyTexts() returns newest-first. Index 0 = most recent.
        // The buffer wraps, so we need to compute the real position.
        size_t bufIdx;
        if (count_ < capacity_) {
            // Not wrapped: indices 0..count_-1 map to buffer positions 0..count_-1
            bufIdx = index;
        } else {
            // Wrapped: newest is at (head_-1) % cap, oldest at head_
            bufIdx = (head_ - 1 - index + capacity_) % capacity_;
        }
        
        const auto &data = buffer_[bufIdx];
        
        // Validate PC
        auto &si = impl.scriptInterpreter;
        if (data.programCounter < si.program.size()) {
            si.pc = data.programCounter;
        } else {
            return false;  // invalid PC
        }
        
        // Restore call stack (with bounds check)
        si.callStack = data.callStack;
        if (si.callStack.size() > 32)
            si.callStack.resize(32);
        
        // Restore variables
        si.variables = data.variables;
        si.numVariables = data.numVariables;
        si.skipMode = data.skipMode;
        si.skipDepth = data.skipDepth;
        si.scriptFinished = false;
        
        // Restore scene
        if (!data.background.empty())
            impl.scene.ShowBackground(data.background);
        else
            impl.scene.Clear();
        
        // Restore characters
        for (const auto &ch : data.characters)
            impl.scene.ShowCharacter(ch.id, ch.file, ch.position);
        
        // Restore audio
        if (!data.bgm.empty())
            impl.audio.PlayBGM(data.bgm);
        
        // Restore dialogue
        impl.dialogue.SetText(data.text);
        impl.dialogue.SetDisplayedChars(data.displayedChars);
        if (!data.speaker.empty()) {
            impl.dialogue.SetSpeaker(data.speaker);
            impl.dialogue.SetName(data.name);
        } else {
            impl.dialogue.Clear();
        }
        
        // Restore state machine — clear overlays and set to restored state
        impl.m_stateMachine.clearOverlays();
        // Parse the saved state string back to CerekaState
        CerekaState savedState = CerekaState::Running;  // default
        std::string st = data.state;
        if (st == "WaitingForInput") savedState = CerekaState::WaitingForInput;
        else if (st == "InMenu") savedState = CerekaState::InMenu;
        else if (st == "Fading") savedState = CerekaState::Fading;
        else if (st == "Finished") savedState = CerekaState::Finished;
        impl.m_stateMachine.changeState(savedState);
        
        return true;
    }
    
    std::vector<std::string> RollbackManager::historyTexts() const {
        std::vector<std::string> texts;
        texts.reserve(count_);
        
        if (count_ < capacity_) {
            for (size_t i = 0; i < count_; ++i)
                texts.push_back(dialogueTexts_[i]);
        } else {
            // Wrapped: start at head_, wrap around
            for (size_t i = 0; i < capacity_; ++i) {
                size_t idx = (head_ + i) % capacity_;
                texts.push_back(dialogueTexts_[idx]);
            }
        }
        // Reverse to show newest first
        std::reverse(texts.begin(), texts.end());
        return texts;
    }
    
    size_t RollbackManager::prevIndex() const {
        if (count_ == 0) return 0;
        return (head_ - 1 + capacity_) % capacity_;
    }
    
    }  // namespace cereka
    ```
