    struct SceneNode {
        std::string id;
        struct Transform {
            float x = 0.5f, y = 0.5f;      // normalized 0-1, center default
            float scaleX = 1.0f, scaleY = 1.0f;
            float rotationDeg = 0.0f;       // degrees clockwise
            float opacity = 1.0f;           // 0.0-1.0
        } local;
        Transform world;  // accumulated, cached each frame in updateTransforms()
        std::shared_ptr<ITexture> texture;  // optional — null until scene_graph sets it, or set later
        bool visible = true;
        SceneNode *parent = nullptr;
        std::vector<std::unique_ptr<SceneNode>> children;
    };

    class SceneGraph {
    public:
        SceneNode *createNode(const std::string &id);
        void removeNode(const std::string &id);
        SceneNode *findNode(const std::string &id);
        bool hasNode(const std::string &id) const;

        // Depth-first: compute world transforms by accumulating local through parent chain.
        // Call once per frame before draw.
        void updateTransforms();

        // Visit every visible node depth-first (for drawing). Passes const SceneNode&.
        void visit(const std::function<void(const SceneNode&)>& visitor) const;
        void setTransform(const std::string &id, const std::string &propStr);

        void Clear();
    private:
        std::unique_ptr<SceneNode> root_;  // hidden root, all user nodes attach here
        std::unordered_map<std::string, SceneNode*> nodeMap_;
        void updateNode(SceneNode &node, const Transform &parentAccum);
        void visitNode(const SceneNode &node, const std::function<void(const SceneNode&)>& visitor) const;
    };

    // In scene_graph.cpp:
    // SceneGraph::setTransform parses the propStr format:
    //   "pos(50%,80%) scale(1.2) rotate(15) opacity(0.8)"
    // Each token sequence: keyword + parenthesized value(s).
    // Use safe_stof for numeric parsing (from cereka_safe_parse.hpp).
    // If parsing fails for a property, leave it at its current/default value (graceful).
    // pos() takes 2 values (x y), either normalized "50%" or pixel "500".
    //   - "%" suffix → divide by 100 for normalized 0-1
    //   - No suffix → divide by screen dimension (passed separately, or store as-is and UIManager normalizes)
    //   For now, store as normalized float directly (0-1): pos(0.5,0.8) means 50%,80%.
    // scale() takes 1 value (uniform) or 2 (scaleX, scaleY): scale(1.5) or scale(1.5,2.0)
    // rotate() takes 1 value in degrees
    // opacity() takes 1 value 0.0-1.0
    //
    // updateTransforms(): depth-first from root_. Compute world = parent.world * local.
    // Multiplication: world.x = parent.world.x + local.x (offset, normalized), world.scaleX = parent.world.scaleX * local.scaleX, etc.
    // For simplicity: world.x = local.x (absolute, not offset from parent). Scale accumulates: world.scaleX = parent.world.scaleX * local.scaleX. Rotation accumulates: world.rotationDeg = parent.world.rotationDeg + local.rotationDeg. Opacity multiplies: world.opacity = parent.world.opacity * local.opacity.
    //
    // Clear(): removes all user nodes, resets root_.

    ```lua
    local function parse_scene_graph(ctx)
        -- scene_graph <id> set <prop-changes>
        -- scene_graph <id> remove
        local kw = take(ctx)
        local id_t = expect(ctx, "IDENT", nil, "expected node id after 'scene_graph'")
        local action_t = peek(ctx)
        if not action_t then die(kw.lineno, kw.col, "expected 'set' or 'remove'") end
        if action_t.type == "IDENT" and action_t.value == "remove" then
            take(ctx)
            return { kind = "SceneGraphRemove", id = id_t.value, line = kw.lineno, col = kw.col }
        elseif action_t.type == "IDENT" and action_t.value == "set" then
            take(ctx)
            local rest = rest_text(ctx)
            if rest == "" then die(kw.lineno, kw.col, "expected property changes after 'set'") end
            return { kind = "SceneGraphSet", id = id_t.value, props = rest, line = kw.lineno, col = kw.col }
        else
            die(action_t.lineno, action_t.col, "expected 'set' or 'remove'")
        end
    end
    ```

    ```cpp
    void UIManager::DrawSceneGraph()
    {
        m_sceneGraph.updateTransforms();
        m_renderCtx->SetBlendMode(true);
        m_sceneGraph.visit([this](const SceneNode &node) {
            if (!node.texture) return;
            // Compute screen-space position from normalized coords
            int sw = m_renderCtx->Width();
            int sh = m_renderCtx->Height();
            float sx = node.world.x * sw;
            float sy = node.world.y * sh;
            float w = node.texture->Width() * node.world.scaleX;
            float h = node.texture->Height() * node.world.scaleY;
            Rect dst{sx - w/2.0f, sy - h/2.0f, w, h};
            // Apply opacity via blend mode (opacity < 1.0 needs alpha blending)
            // For now, draw at full alpha — IRenderContext doesn't have per-texture alpha yet.
            // Future: extend IRenderContext::DrawTexture to accept an alpha param.
            m_renderCtx->DrawTexture(*node.texture, nullptr, &dst);
        });
    }
    ```
    Note: Per-texture alpha is not yet supported by IRenderContext::DrawTexture. The current implementation draws at full opacity. This is sufficient for Phase 4 launch — opacity support can be added to the DrawTexture method in a future refinement. The world.opacity value is computed and cached for when that support arrives.
