    ```cpp
    #pragma once
    #include "renderer/irender_context.hpp"  // for Color
    #include <string>

    namespace cereka::text {

    struct TextStyle {
        bool bold = false;
        bool italic = false;
        bool underline = false;
        bool strikethrough = false;
        Color color = {255, 255, 255, 255};
    };

    struct TextSegment {
        std::string text;
        TextStyle style;
    };

    // Parse angle-bracket markup into style-tagged segments.
    // Tags: <b>, </b>, <i>, </i>, <u>, </u>, <s>, </s>, <color=#rrggbb>, </color>
    // Escaping: << → '<', >> → '>'
    // Returns empty vector if the input is empty.
    // If tags are malformed (e.g. unclosed), the parser closes them at end-of-string gracefully.
    std::vector<TextSegment> ParseMarkup(const std::string &input);

    }  // namespace cereka::text
    ```

    **src/text/markup_parser.cpp:**
    Implement `ParseMarkup()` with a tag-stack approach:
    - Walk through input character by character.
    - `<<` → emit literal '<', advance 2.
    - `>>` → emit literal '>', advance 2.
    - `<` followed by a known tag name → start a style change.
      - If next chars match `b>`, `i>`, `u>`, `s>`, `color=...>`: open a style delta.
      - If next chars match `/b>`, `/i>`, `/u>`, `/s>`, `/color>`: close the corresponding tag from the stack.
    - Any other `<` → treat as literal text character (graceful).
    - Maintain a `std::vector<TextStyle>` stack. When opening a tag, push a copy of current style with the delta applied. When closing, pop the stack.
    - When style changes (push or pop), flush the accumulated text buffer as a TextSegment with the previous style, start a new segment.
    - At end of input, flush remaining text buffer.
    - On unclosed tags at EOF, close them in reverse order (pop stack) to get final segment with base style.

    Color parsing: `<color=#rrggbb>` — extract hex digits after `#`, parse pairs with `std::stoul` (base 16). Invalid hex → leave color at default white. Also support `<color=red>` for named colors later — for now, only hex format is required.

    **src/text/rich_text_renderer.cpp:**
    Implement `DrawRichText()`:
    - This is a standalone function (not on IRenderContext yet — Task 2 will integrate it).
    - How it works:
      1. Line-layout pass: Build a flat list of "glyph runs" — one per (segment, line). For each segment, measure its width via `TTF_MeasureString(font, text, text.length(), &measuredWidth, &extentCount)`. Track accumulated x position. When x + measuredWidth > maxWidth, break to new line, reset x.
      2. If a single segment spans a line break, split the text into "before break" and "after break" sub-segments.
      3. Render pass: For each line, for each segment in the line:
         a. `TTF_SetFontStyle(font, styleBits)` where styleBits = (bold?TTF_STYLE_BOLD:0) | (italic?TTF_STYLE_ITALIC:0) | (underline?TTF_STYLE_UNDERLINE:0) | (strikethrough?TTF_STYLE_STRIKETHROUGH:0)
         b. `SDL_Surface *surf = TTF_RenderText_Blended(font, text.c_str(), text.size(), sdlColor)`
         c. `SDL_Texture *tex = SDL_CreateTextureFromSurface(renderer, surf)`
         d. Create SdlTexture wrapper, draw to IRenderContext at computed (x, y) position.
         e. Advance y by line height (TTF_GetFontHeight or measure from surface height).
    - Note: This implementation directly creates SDL textures. Once merged, the IRenderContext::DrawRichText method (Task 2) will encapsulate this.

    ```cpp
    float SdlRenderContext::DrawRichText(
        TTF_Font *font,
        const std::vector<text::TextSegment> &segments,
        float x, float y, float maxWidth)
    {
        if (segments.empty() || !font) return 0.0f;

        // --- Line layout: accumulate segment widths, break at maxWidth ---
        struct PlacedSegment {
            const text::TextSegment *seg;
            size_t startOffset;  // character offset into seg->text (0 for full segment, >0 for wrapped)
            size_t endOffset;    // exclusive end
            float lineX;
            float lineY;
        };
        std::vector<PlacedSegment> placed;
        float currentX = x;
        float currentY = y;
        float lineHeight = (float)TTF_GetFontHeight(font);

        for (const auto &seg : segments) {
            const char *text = seg.text.c_str();
            size_t textLen = seg.text.size();
            size_t offset = 0;

            while (offset < textLen) {
                // Measure remaining text
                int measuredWidth = 0;
                int extent = 0;
                if (!TTF_MeasureString(font, text + offset, textLen - offset,
                                       (int)(maxWidth - (currentX - x)), &measuredWidth, &extent))
                    break;  // measurement failed

                if (extent == 0) {
                    // Word doesn't fit — wrap to next line
                    currentX = x;
                    currentY += lineHeight;
                    continue;
                }

                placed.push_back({&seg, offset, offset + (size_t)extent, currentX, currentY});
                currentX += (float)measuredWidth;
                offset += (size_t)extent;
            }
        }

        if (placed.empty()) return 0.0f;

        // --- Render each placed segment ---
        for (const auto &ps : placed) {
            int style = 0;
            if (ps.seg->style.bold)          style |= TTF_STYLE_BOLD;
            if (ps.seg->style.italic)        style |= TTF_STYLE_ITALIC;
            if (ps.seg->style.underline)     style |= TTF_STYLE_UNDERLINE;
            if (ps.seg->style.strikethrough) style |= TTF_STYLE_STRIKETHROUGH;
            TTF_SetFontStyle(font, style);

            SDL_Color sdlColor{
                ps.seg->style.color.r,
                ps.seg->style.color.g,
                ps.seg->style.color.b,
                ps.seg->style.color.a
            };

            std::string subText = ps.seg->text.substr(ps.startOffset, ps.endOffset - ps.startOffset);
            SDL_Surface *surf = TTF_RenderText_Blended(
                font, subText.c_str(), subText.size(), sdlColor);
            if (!surf) continue;

            SDL_Texture *tex = SDL_CreateTextureFromSurface(m_renderer, surf);
            float tw = (float)surf->w;
            float th = (float)surf->h;
            SDL_DestroySurface(surf);

            if (tex) {
                SDL_FRect dst{ps.lineX, ps.lineY, tw, th};
                SDL_RenderTexture(m_renderer, tex, nullptr, &dst);
                SDL_DestroyTexture(tex);
            }
        }

        // Return total height
        return (currentY - y) + lineHeight;
    }
    ```
