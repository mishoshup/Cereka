```cpp
if (plat.name == "linux") {
    fs::permissions(stagingDir / gameExe,
                    fs::perms::owner_exec | fs::perms::group_exec |
                        fs::perms::others_exec,
                    fs::perm_options::add, ec);
    // Bundle SDL shared libraries alongside the binary.
    // The binary's $ORIGIN RPATH (set in runner/CMakeLists.txt) resolves
    // .so files from the same directory — so they must be in the ZIP.
    static const char* soLibs[] = {
        "libSDL3.so.0", "libSDL3_image.so.0",
        "libSDL3_ttf.so.0", "libSDL3_mixer.so.0"
    };
    for (auto* lib : soLibs) {
        fs::path src = runtimeDir("linux") / lib;
        if (fs::exists(src)) {
            fs::copy_file(src, stagingDir / lib,
                          fs::copy_options::overwrite_existing, ec);
            if (ec)
                appendLog("[ERROR] Cannot copy " + std::string(lib) + ": " + ec.message());
            else
                appendLog("Copied " + std::string(lib));
        } else {
            appendLog("[WARN] Not found, skipping: " + std::string(lib));
        }
    }
    QMetaObject::invokeMethod(this, [this]() { updateLog(); }, Qt::QueuedConnection);
} else {
```

Key decisions in this implementation:
- Use a named-list `soLibs[]` instead of directory iteration — SDL .so files have known sonames; directory iteration risks picking up `.so` dev symlinks or build artifacts
- Use `fs::exists(src)` check before copying — if Plan 01's CMake routing is not yet applied (separate build), the packager warns instead of hard-failing
- Use `[WARN]` (not `[ERROR]`) for missing .so files so the packager continues and produces the rest of the archive — the developer sees the warning and can investigate
- Use the `std::error_code` overload (not the throwing overload) — consistent with all other copy calls in this function
- Single `QMetaObject::invokeMethod` flush after the loop — the loop may copy 4 files quickly; one flush at the end is sufficient
